class RECT(ctypes.Structure):
    _fields_ = [
        ("left", ctypes.c_int),
        ("top", ctypes.c_int),
        ("right", ctypes.c_int),
        ("bottom", ctypes.c_int),
    ]


def get_work_area() -> tuple[int, int, int, int]:
    r = RECT()
    ctypes.windll.user32.SystemParametersInfoW(SPI_GETWORKAREA, 0, ctypes.byref(r), 0)
    return r.left, r.top, r.right, r.bottom


def get_screen_size() -> tuple[int, int]:
    return (
        ctypes.windll.user32.GetSystemMetrics(0),  # SM_CXSCREEN
        ctypes.windll.user32.GetSystemMetrics(1),  # SM_CYSCREEN
    )


def taskbar_band_height() -> int:
    """Pixels from work-area bottom to screen bottom (typ. taskbar height)."""
    _, _, _, wa_bottom = get_work_area()
    _, cy = get_screen_size()
    return max(0, cy - wa_bottom)


def sample_taskbar_center_pixel_rgb() -> tuple[int, int, int] | None:
    """
    Sample RGB at the center of the bottom taskbar band (Windows).
    Used to match the strip to the user's taskbar, then we darken slightly for contrast with glow.
    """
    if sys.platform != "win32":
        return None
    try:
        left, _top, right, wa_bottom = get_work_area()
        _cx, cy = get_screen_size()
        tb_h = cy - wa_bottom
        if tb_h < 8:
            return None
        px = (left + right) // 2
        py = wa_bottom + max(2, min(tb_h // 2, tb_h - 2))
        hdc = ctypes.windll.user32.GetDC(0)
        if not hdc:
            return None
        c = ctypes.windll.gdi32.GetPixel(hdc, px, py)
        ctypes.windll.user32.ReleaseDC(0, hdc)
        if c == -1 or c == 0xFFFFFFFF:
            return None
        b = c & 0xFF
        g = (c >> 8) & 0xFF
        r = (c >> 16) & 0xFF
        return (r, g, b)
    except (OSError, AttributeError, ctypes.ArgumentError):
        return None


def win32_foreground_pid() -> int | None:
    """PID of the foreground window’s process (Windows)."""
    if sys.platform != "win32":
        return None
    try:
        hwnd = ctypes.windll.user32.GetForegroundWindow()
        if not hwnd:
            return None
        pid = ctypes.c_ulong()
        ctypes.windll.user32.GetWindowThreadProcessId(hwnd, ctypes.byref(pid))
        return int(pid.value)
    except (OSError, AttributeError, ctypes.ArgumentError, ValueError):
        return None


# Subtract foreground process CPU from system average for alert math only (display stays raw).
ALERT_SUBTRACT_FOREGROUND_CPU = True


def cpu_load_for_alerts(
    cpu_load_01: float,
    fg_cpu_ema: float,
    ncpu: int,
) -> float:
    """
    CPU load in [0,1] comparable to cpu_load_01, with foreground process contribution removed
    for warning / maxed detection. Uses the same capacity scale: both total average and foreground
    are expressed as a fraction of (100% × logical CPUs).
    """
    if not ALERT_SUBTRACT_FOREGROUND_CPU or fg_cpu_ema <= 0:
        return cpu_load_01
    cap = 100.0 * max(ncpu, 1)
    fg_as_fraction_of_capacity = min(1.0, max(0.0, fg_cpu_ema) / max(cap, 1e-9))
    return max(0.0, cpu_load_01 - fg_as_fraction_of_capacity)


def maxed_frac_for_channel(idx: int) -> float:
    """Sample counts as 'maxed' for minute-red when alert load ≥ this (per-channel pegs)."""
    if idx == 0:
        return MAXED_FRAC_CPU
    if idx in (1, 3):  # GPU, VRAM
        return MAXED_FRAC_GPU_VRAM
    if idx in (2, 4, 5):  # RAM, DISK, SWAP
        return MAXED_FRAC_MEM_PRESSURE
    if idx == 6:
        return MAXED_FRAC_NET
    return MAXED_FRAC


def minute_hot_fraction_for_channel(idx: int) -> float:
    """Fraction of minute-window samples that must be maxed to arm a red card for this channel."""
    if idx == 0:
        return MINUTE_HOT_FRACTION_CPU
    if idx in (1, 3):
        return MINUTE_HOT_FRACTION_GPU_VRAM
    if idx in (2, 4, 5):
        return MINUTE_HOT_FRACTION_MEM_PRESSURE
    if idx == 6:
        return MINUTE_HOT_FRACTION_NET
    return MINUTE_HOT_FRACTION

    """

def rgb_hex_darkened(r: int, g: int, b: int, factor: float = 0.9) -> str:
    """Darken RGB by ~10% (factor 0.9) and return #rrggbb."""
    r = max(0, min(255, int(r * factor)))
    g = max(0, min(255, int(g * factor)))
    b = max(0, min(255, int(b * factor)))
    return f"#{r:02x}{g:02x}{b:02x}"

def win32_hwnd_topmost(hwnd: int) -> None:
    if sys.platform != "win32" or not hwnd:
        return
    try:
        ctypes.windll.user32.SetWindowPos(
            hwnd,
            HWND_TOPMOST,
            0,
            0,
            0,
            0,
            SWP_NOMOVE | SWP_NOSIZE | SWP_SHOWWINDOW | SWP_NOACTIVATE,
        )
    except (OSError, AttributeError, ctypes.ArgumentError, ValueError, TypeError):
        pass


def load_meter_config() -> dict:
    if not CONFIG_PATH.is_file():
        return {}
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, dict) else {}
    except (OSError, json.JSONDecodeError):
        return {}


def save_meter_config(data: dict) -> None:
    try:
        with open(CONFIG_PATH, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)
    except OSError:
        pass


def _try_nvidia_gpu_vram() -> tuple[float, float, str]:
    """
    Returns (gpu_util_01, vram_used_frac_01, detail_line) using nvidia-smi when available.
    """
    if sys.platform == "win32":
        creationflags = getattr(subprocess, "CREATE_NO_WINDOW", 0)
    else:
        creationflags = 0
    try:
        p = subprocess.run(
            [
                "nvidia-smi",
                "--query-gpu=utilization.gpu,memory.used,memory.total",
                "--format=csv,noheader,nounits",
            ],
            capture_output=True,
            text=True,
            timeout=1.8,
            creationflags=creationflags,
        )
        if p.returncode != 0 or not (p.stdout or "").strip():
            return 0.0, 0.0, ""
        line = p.stdout.strip().splitlines()[0]
        parts = [x.strip() for x in line.split(",")]
        if len(parts) < 3:
            return 0.0, 0.0, ""
        gpu = max(0.0, min(100.0, float(parts[0]))) / 100.0
        mused = float(parts[1])
        mtot = float(parts[2])
        vram = mused / max(mtot, 1.0)
        return min(1.0, gpu), min(1.0, vram), f"{parts[0].strip()}% util, {parts[1]}/{parts[2]} MiB"
    except (OSError, subprocess.TimeoutExpired, ValueError, IndexError):
        return 0.0, 0.0, ""


class RollingStatsBuffer:
    """Rolling window for 5-second median, min, max, spread (for band sway / display)."""

    def __init__(self, window_sec: float = MEDIAN_WINDOW_SEC) -> None:
        self._window = window_sec
        self._samples: list[tuple[float, float]] = []

    def push(self, t: float, v: float) -> None:
        self._samples.append((t, v))
        cutoff = t - self._window
        self._samples = [(t0, v0) for t0, v0 in self._samples if t0 >= cutoff]

    def median(self) -> float | None:
        if not self._samples:
            return None
        vals = [v for _, v in self._samples]
        return float(statistics.median(vals))

    def min(self) -> float | None:
        if not self._samples:
            return None
        return min(v for _, v in self._samples)

    def max(self) -> float | None:
        if not self._samples:
            return None
        return max(v for _, v in self._samples)

    def spread(self) -> float:
        if len(self._samples) < 2:
            return 0.0
        lo = self.min()
        hi = self.max()
        if lo is None or hi is None:
            return 0.0
        return max(0.0, hi - lo)

    @property
    def sample_count(self) -> int:
        return len(self._samples)


class MinuteHotTracker:
    """Tracks what fraction of the rolling window (see MINUTE_WINDOW_SEC) was 'maxed'."""

    def __init__(self, window_sec: float = MINUTE_WINDOW_SEC) -> None:
        self._window = window_sec
        self._samples: list[tuple[float, bool]] = []

    def push(self, t: float, maxed: bool) -> None:
        self._samples.append((t, maxed))
        cutoff = t - self._window
        self._samples = [(t0, v0) for t0, v0 in self._samples if t0 >= cutoff]

    def fraction_maxed(self) -> float:
        if not self._samples:
            return 0.0
        return sum(1 for _, v in self._samples if v) / len(self._samples)

    @property
    def sample_count(self) -> int:
        return len(self._samples)


class Metrics:
    def __init__(self) -> None:
        self._prev_net = psutil.net_io_counters()
        self._prev_disk = psutil.disk_io_counters()
        self._prev_t = time.perf_counter()
        self._boot = True
        self._net_cap_bps = 125_000_000
        self._disk_rw_cap_bps = 800_000_000
        self._gpu_cache_t = 0.0
        self._gpu_u = 0.0
        self._vram_u = 0.0
        self._gpu_detail = ""

    def _gpu_vram(self) -> tuple[float, float, str]:
        now = time.monotonic()
        if now - self._gpu_cache_t < 2.0 and self._gpu_cache_t > 0:
            return self._gpu_u, self._vram_u, self._gpu_detail
        self._gpu_cache_t = now
        g, v, d = _try_nvidia_gpu_vram()
        self._gpu_u, self._vram_u, self._gpu_detail = g, v, d
        return g, v, d

    def sample(self) -> tuple[list[tuple[float, str]], dict]:
        now = time.perf_counter()
        dt = max(now - self._prev_t, 0.05)
        self._prev_t = now

        percpu = psutil.cpu_percent(interval=None, percpu=True)
        ncpu = len(percpu) if percpu else psutil.cpu_count() or 1
        if not percpu:
            percpu = [psutil.cpu_percent(interval=None)]
        agg = float(sum(percpu))
        cpu_load_01 = min(agg / (100.0 * max(ncpu, 1)), 1.0)

        vm = psutil.virtual_memory()
        ram = vm.percent / 100.0

        gpu_u, vram_u, gpu_line = self._gpu_vram()
        if not gpu_line:
            gpu_line = "GPU (nvidia-smi not available or no NVIDIA GPU)"

        sw = psutil.swap_memory()
        swap_u = 0.0 if sw.total == 0 else sw.percent / 100.0

        net = psutil.net_io_counters()
        d_net = (net.bytes_sent + net.bytes_recv) - (
            self._prev_net.bytes_sent + self._prev_net.bytes_recv
        )
        self._prev_net = net
        net_bps = max(d_net, 0) / dt
        net_u = min(net_bps / self._net_cap_bps, 1.0)

        dsk = psutil.disk_io_counters()
        read_bps = write_bps = 0.0
        if dsk and self._prev_disk:
            read_bps = max(dsk.read_bytes - self._prev_disk.read_bytes, 0) / dt
            write_bps = max(dsk.write_bytes - self._prev_disk.write_bytes, 0) / dt
        self._prev_disk = dsk

        rw_bps = read_bps + write_bps
        disk_io_u = min(rw_bps / self._disk_rw_cap_bps, 1.0)

        if self._boot:
            self._boot = False
            net_u = disk_io_u = 0.0

        avg_pct = agg / max(ncpu, 1)
        tip_cpu = (
            f"CPU (Task Manager style)\n"
            f"{agg:.0f}% aggregate  ({ncpu} logical)\n"
            f"~{avg_pct:.0f}% average across cores\n"
            f"Lumax alerts if this stays high for a long time."
        )
        tip_gpu = f"GPU\n{gpu_line}"
        tip_ram = (
            f"RAM\n{vm.percent:.0f}% used\n"
            f"{vm.used // (1024**3)} / {vm.total // (1024**3)} GiB"
        )
        tip_vram = (
            f"VRAM\n{vram_u * 100:.0f}% of reported VRAM (NVIDIA when available)"
        )
        tip_disk = (
            f"Disk I/O (read+write)\n{rw_bps / 1e6:.1f} MB/s total\n"
            f"(read {read_bps / 1e6:.1f}, write {write_bps / 1e6:.1f})"
        )
        tip_swap = f"Swap\n{sw.percent:.0f}% used" if sw.total > 0 else "Swap\nno swap configured"
        tip_net = (
            f"Network\n{net_bps / 1e6:.1f} MB/s in+out\n"
            f"(~{self._net_cap_bps // 1_000_000} MB/s cap for bar)"
        )

        lines = [
            (cpu_load_01, tip_cpu),
            (gpu_u, tip_gpu),
            (ram, tip_ram),
            (vram_u, tip_vram),
            (disk_io_u, tip_disk),
            (swap_u, tip_swap),
            (net_u, tip_net),
        ]
        shorts = [
            f"{agg:.0f}%",
            f"{gpu_u * 100:.0f}%",
            f"{vm.percent:.0f}%",
            f"{vram_u * 100:.0f}%",
            f"{rw_bps / 1e6:.1f} MB/s",
            f"{sw.percent:.0f}%",
            f"{net_bps / 1e6:.1f} MB/s",
        ]
        meta = {
            "cpu_agg": agg,
            "ncpu": ncpu,
            "cpu_load_01": cpu_load_01,
        }
        return lines, meta, shorts


class Tooltip:
    def __init__(self, widget: tk.Widget) -> None:
        self._w = widget
        self._tip: tk.Toplevel | None = None

    def show(self, text: str, x_root: int, y_root: int) -> None:
        self.hide()
        tw = tk.Toplevel(self._w)
        self._tip = tw
        tw.wm_overrideredirect(True)
        tw.wm_attributes("-topmost", True)
        if sys.platform == "win32":
            tw.wm_attributes("-alpha", 0.94)
        lbl = tk.Label(
            tw,
            text=text,
            justify=tk.LEFT,
            background="#0d1118",
            foreground="#c8f4ff",
            font=("Consolas", 9),
            padx=10,
            pady=8,
            highlightthickness=1,
            highlightbackground="#00f3ff",
        )
        lbl.pack()
        tw.update_idletasks()
        h = tw.winfo_reqheight()
        tw.wm_geometry(f"+{x_root + 14}+{y_root - h - 8}")

    def hide(self) -> None:
        if self._tip:
            try:
                self._tip.destroy()
            except tk.TclError:
                pass
            self._tip = None

    def _chrome_bg(self) -> str:
        if self._transparent:
            return self.TRANSPARENT_MAGIC
        return self._meter_surface_bg()

    def _highlight_base(self) -> str:
        if self._style == "flow":
            return "#5a7080" if self._transparent else self.BORDER_IDLE
        if self._style in ("rhomb", "arrows"):
            return self.BORDER_IDLE
        return "#333333"

    def _apply_window_visual(self) -> None:
        """Transparent: Windows chroma-key; elsewhere, slightly faded alpha."""
        if self._transparent:
            if sys.platform == "win32":
                try:
                    self.attributes("-transparentcolor", self.TRANSPARENT_MAGIC)
                except tk.TclError:
                    pass
                self.attributes("-alpha", 1.0)
            else:
                try:
                    self.attributes("-transparentcolor", "")
                except tk.TclError:
                    pass
                self.attributes("-alpha", 0.88)
        else:
            if sys.platform == "win32":
                try:
                    self.attributes("-transparentcolor", "")
                except tk.TclError:
                    pass
            self.attributes("-alpha", self._alpha)

    def _ping_win32_topmost(self) -> None:
        if not self._force_win_topmost:
            return
        try:
            self.update_idletasks()
            hwnd = int(self.winfo_id())
            win32_hwnd_topmost(hwnd)
        except (tk.TclError, ValueError, TypeError, OverflowError):
            pass

    def _on_focus_out(self, event: tk.Event) -> None:
        if event.widget not in (self, self.canvas):
            return
        self._ping_win32_topmost()

    def _on_map(self, _event: tk.Event) -> None:
        self._ping_win32_topmost()

    def _topmost_loop(self) -> None:
        try:
            if not self.winfo_exists():
                return
        except tk.TclError:
            return
        if self._sticky_topmost:
            try:
                self.attributes("-topmost", True)
                self.lift()
            except tk.TclError:
                pass
        if self._force_win_topmost:
            self._ping_win32_topmost()
        self.after(1500, self._topmost_loop)

    def _layout_metrics(self) -> None:
        gap = 5
        tb_h = taskbar_band_height()
        if self._appbar:
            self._ww, self._hh = 220, 40
            self._pad = 6
            self._seg_w = 34
            self._seg_h = 26
            self._plate_bg = self.PLATE_BG
            self._alpha = 0.96
        elif self._merge_taskbar and tb_h > 8:
            self._pad = 5
            self._seg_w = 22
            self._seg_h = max(14, min(tb_h - 8, 20))
            self._hh = max(tb_h - 4, self._seg_h + 8)
            self._ww = (
                self._pad * 2
                + N_METRICS * self._seg_w
                + (N_METRICS - 1) * gap
                + 4
            )
            self._plate_bg = self.PLATE_BG
            self._alpha = 0.96
        elif self._style == "flow":
            self._ww, self._hh = 320, 34
            self._pad = 6
            self._flow_bar_h = 12
            self._plate_bg = self.BG
            self._alpha = 1.0 if self._transparent else 0.92
        elif self._style in PLATE_STYLES:
            self._pad = 8
            self._seg_w = 28
            self._seg_h = 22
            self._hh = 44 if self._style == "stripe" else 42
            self._ww = (
                self._pad * 2
                + N_METRICS * self._seg_w
                + (N_METRICS - 1) * gap
                + 8
            )
            self._plate_bg = self.PLATE_BG
            self._alpha = 0.94
        else:
            self._pad = 6
            self._seg_w = 28
            self._seg_h = 28
            self._ww = (
                self._pad * 2
                + N_METRICS * self._seg_w
                + (N_METRICS - 1) * gap
            )
            self._hh = 44
            self._plate_bg = self.BG
            self._alpha = 0.92

        self._taskbar_rgb_base = None
        if sys.platform == "win32" and (
            self._appbar or self._merge_taskbar or self._style in PLATE_STYLES
        ):
            t = sample_taskbar_center_pixel_rgb()
            if t is not None:
                self._taskbar_rgb_base = t
        if self._taskbar_rgb_base is not None:
            r, g, b = self._taskbar_rgb_base
            self._plate_bg = rgb_hex_darkened(r, g, b, self._darken_factor)

    def _apply_plate_bgs_to_chrome(self) -> None:
        try:
            bg = self._chrome_bg()
            self.configure(bg=bg)
            if hasattr(self, "canvas") and self.canvas.winfo_exists():
                self.canvas.configure(bg=bg)
                try:
                    self.canvas.itemconfigure("plate_backdrop", fill=bg)
                except tk.TclError:

    def _meter_surface_bg(self) -> str:
        if self._style in PLATE_STYLES or self._merge_taskbar or self._appbar:
            return self._plate_bg
        return self.BG

    def _apply_style(self, style: str, *, persist: bool = True) -> None:
        if style not in STYLES:
            return
        self._style = style
        self._layout_metrics()
        try:
            x, y = self.winfo_x(), self.winfo_y()
        except tk.TclError:
            x, y = 0, 0
        self.configure(bg=self._chrome_bg())
        self._apply_window_visual()
        hl = 2 if not self._merge_taskbar else 1
        hb = self._highlight_base()
        self.canvas.configure(
            width=self._ww,
            height=self._hh,
            bg=self._chrome_bg(),
            highlightthickness=hl,
            highlightbackground=hb,
        )
        self.geometry(f"{self._ww}x{self._hh}+{x}+{y}")
        self._build_segments()
        if persist:
            self._persist_config()

        self._locked = locked
        self.canvas.config(cursor="arrow" if locked else "fleur")
        if not locked:
            self._pulse_toggle = False
        if save:
            self._persist_config()

    def _persist_config(self) -> None:
        save_meter_config(
            {
                "geometry": self.geometry(),
                "locked": self._locked,
                "appbar": self._appbar,
                "style": self._style,
                "focus_metric": self._focus_metric,
                "taskbar_darken_factor": self._darken_factor,
                "transparent": self._transparent,
                "sticky_topmost": self._sticky_topmost,
                "force_win_topmost": self._force_win_topmost,
            }
        )

    def _apply_lock_from_config(self) -> None:
        if load_meter_config().get("locked"):
            self._set_locked(True, save=False)

    def _setup_appbar(self) -> None:
        try:
            from appbar_win32 import apply_bottom_appbar_band, cleanup_appbar
        except ImportError:
            self._apply_saved_geometry()
            return
        try:
            self.update()
            x, y, w, h = apply_bottom_appbar_band(self)
            if w != self._ww or h != self._hh:
                self._ww, self._hh = w, h
                self.canvas.configure(width=w, height=h)
                self._build_segments()
            self.geometry(f"{w}x{h}+{x}+{y}")
            self._appbar_active = True
            self._apply_lock_from_config()
        except Exception:
            try:
                cleanup_appbar(self)
            except Exception:
                pass
            self._appbar = False
            self._apply_saved_geometry()

    def _apply_saved_geometry(self) -> None:
        cfg = load_meter_config()
        geo = cfg.get("geometry")
        if isinstance(geo, str) and geo.count("+") >= 2:
            try:
                self.geometry(geo)
            except tk.TclError:
                self._place_window()
        else:
            self._place_window()
        self._apply_lock_from_config()

    def _on_destroy(self, event: tk.Event) -> None:
        if event.widget is not self:
            return
        if self._appbar_active and sys.platform == "win32":
            try:
                from appbar_win32 import cleanup_appbar

                cleanup_appbar(self)
            except Exception:
                pass
        self._persist_config()

    def _rhombus_points(self, cx: float, cy: float, w: float, h: float) -> list[float]:
        skew = w * 0.22
        return [
            cx - w / 2 + skew,
            cy,
            cx + skew,
            cy - h / 2,
            cx + w / 2 + skew,
            cy,
            cx + skew,
            cy + h / 2,
        ]

    def _arrow_points_right(self, cx: float, cy: float, w: float, h: float) -> list[float]:
        """Solid arrowhead pointing right (tip on +x)."""
        return [
            cx + w / 2,
            cy,
            cx - w / 2,
            cy - h / 2,
            cx - w / 2,
            cy + h / 2,
        ]

    @staticmethod
    def _scale_poly_points(pts: list[float], cx: float, cy: float, scale: float) -> list[float]:
        out: list[float] = []
        for i in range(0, len(pts), 2):
            x, y = pts[i], pts[i + 1]
            out.append(cx + (x - cx) * scale)
            out.append(cy + (y - cy) * scale)
        return out

    def _glow_poly_stack(self, pts: list[float], cx: float, cy: float) -> None:
        for sc, gcol in zip(GLOW_SCALES, GLOW_COLORS):
            gpg = self._scale_poly_points(pts, cx, cy, sc)
            self.canvas.create_polygon(
                gpg,
                fill=gcol,
                outline="",
                smooth=False,
            )

    def _glow_rect_stack_inner(
        self,
        ix1: float,
        iy1: float,
        ix2: float,
        iy2: float,
    ) -> None:
        for pad, gcol in zip(GLOW_RECT_PADS, GLOW_COLORS):
            self.canvas.create_rectangle(
                ix1 - pad,
                iy1 - pad,
                ix2 + pad,
                iy2 + pad,

            )

    def _glow_oval_stack_inner(
        self,
        ix1: float,
        iy1: float,

    ) -> None:
        for pad, gcol in zip(GLOW_RECT_PADS, GLOW_COLORS):

                iy1 - pad,
                ix2 + pad,

            )

        if self._hot_popup is not None:

            except tk.TclError:
                pass
            self._hot_popup = None

        if self._hover_seg != idx or not self._hot_alert[idx]:

    def _first_hot_index(self) -> int | None:
        for i in range(N_METRICS):
            if self._hot_alert[i]:
                return i
        return None

    def _update_foreground_cpu_ema(self, meta: dict) -> None:
        """Smooth foreground process CPU% (Windows) to compare with aggregate load."""
        if sys.platform != "win32":
            return
        pid = win32_foreground_pid()
        if pid is None:
            return
        if pid != self._fg_pid_last:
            self._fg_pid_last = pid
            self._fg_cpu_ema = 0.0
        try:
            p = psutil.Process(pid)
            c = float(p.cpu_percent(interval=None))
        except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
            return
        self._fg_cpu_ema = 0.18 * c + 0.82 * self._fg_cpu_ema

    def _suppress_cpu_hot_due_to_foreground_activity(self, meta: dict) -> bool:
        """
        True ⇒ do not show CPU red alert: focused app likely explains sustained high CPU
        (render/game/compile), not a background anomaly.
        """
        if sys.platform != "win32":
            return False
        agg = float(meta.get("cpu_agg", 0.0))
        if agg < FOREGROUND_DOMINANCE_MIN_AGG:
            return False
        if self._fg_cpu_ema <= 1.0:
            return False
        ratio = self._fg_cpu_ema / max(agg, 1.0)
        return ratio >= FOREGROUND_DOMINANCE_RATIO

    def _compute_segment_fill(
        self,
        i: int,
        levels: list[float],
        per_segment: bool,
        u_single: float,
        now: float,
        pulse_on: bool,
    ) -> tuple[str, str]:
        lv = levels[i] if per_segment else u_single
        rs = self._rolling[i]
        if self._hot_alert[i]:
            t = math.sin(now * 3.1) * 0.5 + 0.5
            fill = self._blend("#ff0000", "#ff6060", t)
            outline = "#ff2200"
            return fill, outline
        if per_segment and rs.spread() >= SPREAD_BAND_FRAC and rs.sample_count >= 3:
            lo = rs.min() or 0.0
            hi = rs.max() or 1.0
            t = math.sin(now * 2.3) * 0.5 + 0.5
            va = lo + t * (hi - lo)
            fill = self._fill_for(va)
            outline = self.CYAN if va > 0.15 else "#2a4a5a"
        else:
            fill = self._fill_for(lv)
            outline = self.CYAN if lv > 0.15 else "#2a4a5a"
        if pulse_on and i == 0:
            outline = self.PINK
        return fill, outline

    def _show_hot_metric_popup(self, idx: int) -> None:
        if self._hot_popup is not None:
            try:
                self._hot_popup.destroy()
            except tk.TclError:
                pass
            self._hot_popup = None
        name = METRIC_LABELS[idx]
        tw = tk.Toplevel(self)
        self._hot_popup = tw
        tw.title(f"{name} — high load")
        tw.attributes("-topmost", True)
        tw.configure(bg="#12080c")
        tw.resizable(False, False)

        body = (
            f"{name} has been at or near maximum for most of the last minute.\n\n"
            "That can be normal during heavy work, or a sign something is stuck. "
            "You can inspect live processes in Task Manager."
        )
        frm = tk.Frame(tw, bg="#12080c", padx=14, pady=12)
        frm.pack(fill=tk.BOTH)
        tk.Label(
            frm,
            text=f"{name} sustained high",
            font=("Segoe UI", 11, "bold"),
            fg="#ff6680",
            bg="#12080c",
        ).pack(anchor="w")
        tk.Label(
            frm,
            text=body,
            font=("Segoe UI", 9),
            fg="#e8e0e4",
            bg="#12080c",
            justify=tk.LEFT,
            wraplength=340,
        ).pack(anchor="w", pady=(8, 0))

        def open_tm() -> None:
            try:
                subprocess.Popen(["taskmgr.exe"], shell=False)
            except OSError:
                subprocess.run("taskmgr", shell=True)

        row = tk.Frame(frm, bg="#12080c")
        row.pack(fill=tk.X, pady=(14, 0))
        def close_hot() -> None:
            try:
                tw.destroy()
            except tk.TclError:
                pass
            self._hot_popup = None

        tk.Button(
            row,
            text="Open Task Manager",
            command=lambda: (open_tm(), close_hot()),
            font=("Segoe UI", 9),
            bg="#3a1820",
            fg="#ffc8d0",
            relief=tk.FLAT,
            padx=12,
            pady=6,
        ).pack(side=tk.LEFT)
        tk.Button(
            row,
            text="Dismiss",
            command=close_hot,
            font=("Segoe UI", 9),
            bg="#1a1a22",
            fg="#ccc",
            relief=tk.FLAT,
            padx=10,
            pady=6,
        ).pack(side=tk.LEFT, padx=(8, 0))

        tw.update_idletasks()
        w, h = tw.winfo_reqwidth(), tw.winfo_reqheight()
        px = self.winfo_rootx() + (self._ww - w) // 2
        py = self.winfo_rooty() + (self._hh - h) // 2
        tw.geometry(f"+{max(8, px)}+{max(8, py)}")

    def _on_canvas_leave(self, _e: tk.Event) -> None:
        self._hover_seg = None
        self._close_hot_popup()
        self.after(120, self._verify_pointer_left_window)

    def _on_pointer_enter_widget(self, _e: tk.Event) -> None:
        self._thanks_flash_until = None

    def _on_toplevel_leave(self, _e: tk.Event) -> None:
        self.after(120, self._verify_pointer_left_window)

    def _verify_pointer_left_window(self) -> None:
        try:
            if not self.winfo_exists():
                return
        except tk.TclError:
            return
        px, py = self.winfo_pointerxy()
        x1, y1 = self.winfo_rootx(), self.winfo_rooty()
        x2, y2 = x1 + self.winfo_width(), y1 + self.winfo_height()
        if x1 <= px < x2 and y1 <= py < y2:
            return
        self._start_thanks_flash()

    def _start_thanks_flash(self) -> None:
        self._thanks_flash_until = time.monotonic() + 2.0

    def _on_canvas_motion(self, e: tk.Event) -> None:
        if self._style != "flow":
            return
        ix1, iy1, ix2, iy2 = self._flow_track
        if e.x < ix1 or e.x > ix2 or e.y < iy1 or e.y > iy2:
            self._hover_seg = None
            self._close_hot_popup()
            return
        span = (ix2 - ix1) / float(N_METRICS)
        idx = int((e.x - ix1) / span)
        idx = max(0, min(N_METRICS - 1, idx))
        self._hover_seg = idx
        self._cancel_hot_popup_sched()
        if self._hot_alert[idx]:
            self._hot_popup_sched = self.after(
                280, lambda j=idx: self._open_hot_popup_if(j)
            )
        else:
            self._close_hot_popup()

    def _set_focus_metric(self, idx: int | None) -> None:
        self._focus_metric = idx
        self._persist_config()

    def _apply_focus_from_config(self) -> None:
        cfg = load_meter_config()
        fm = cfg.get("focus_metric")
        if fm is None:
            self._focus_metric = None
        elif isinstance(fm, int) and 0 <= fm < N_METRICS:
            self._focus_metric = fm
        else:

    def _build_grille(self) -> None:
        """Horizontal slats per column — speaker-grille / vent readout."""
        self._plate_backdrop()

            1.5,

    def _fill_for(self, level: float) -> str:
        level = max(0.0, min(1.0, level))
        if level < 0.35:
            return self._blend(self.DIM, self.CYAN, level / 0.35 * 0.5)
        if level < 0.7:
            return self._blend(self.CYAN, self.GREEN, (level - 0.35) / 0.35)
        return self._blend(self.GREEN, self.PINK, (level - 0.7) / 0.3)

    @staticmethod
    def _fill_blue_red(level: float) -> str:
        """Left (low) → blue; right (high) → red."""
        level = max(0.0, min(1.0, level))
        lo = "#2080ff"
        hi = "#ff2028"
        return StressMeter._blend(lo, hi, level)

    @staticmethod
    def _blend(a: str, b: str, t: float) -> str:
        t = max(0.0, min(1.0, t))

        def hx(s: str) -> tuple[int, int, int]:
            s = s.lstrip("#")
            return tuple(int(s[i : i + 2], 16) for i in (0, 2, 4))

        ra, ga, ba = hx(a)
        rb, gb, bb = hx(b)
        r = int(ra + (rb - ra) * t)
        g = int(ga + (gb - ga) * t)
        b = int(ba + (bb - ba) * t)
        return f"#{r:02x}{g:02x}{b:02x}"

    def _update_cpu_sustained(self, meta: dict) -> None:
        load = float(meta.get("cpu_load_01", 0.0))
        now = time.monotonic()
        if load >= CPU_HOT_FRAC:
            if self._cpu_hot_since is None:
                self._cpu_hot_since = now
            sustained = now - self._cpu_hot_since
            if sustained >= FIRST_ALERT_AFTER_SEC:
                may_alert = self._last_cpu_alert_at is None or (
                    now - self._last_cpu_alert_at >= REPEAT_ALERT_SEC
                )
                if may_alert and self._alert_window is None:
                    self._show_caring_cpu_alert(meta, sustained)
                    self._last_cpu_alert_at = now
        elif load <= CPU_COOL_FRAC:
            self._cpu_hot_since = None

    def _tick(self) -> None:
        rows, meta, shorts = self._metrics.sample()
        self._update_cpu_sustained(meta)

        sustained = 0.0
        if self._cpu_hot_since is not None:
            sustained = time.monotonic() - self._cpu_hot_since

        pulse_on = sustained >= PULSE_BORDER_AFTER_SEC and meta.get("cpu_load_01", 0) >= CPU_HOT_FRAC
        idle_border = self._idle_border_color()
        hot_border = self.BORDER_HOT
        if pulse_on:
            self._pulse_toggle = not self._pulse_toggle
            self.canvas.configure(
                highlightbackground=hot_border if self._pulse_toggle else idle_border
            )
        else:
            self.canvas.configure(highlightbackground=idle_border)

        levels = [r[0] for r in rows]
        now = time.monotonic()
        self._update_foreground_cpu_ema(meta)
        ncpu = int(meta.get("ncpu", 1))
        cpu_for_alerts = cpu_load_for_alerts(levels[0], self._fg_cpu_ema, ncpu)
        for i, row in enumerate(rows):
            self._rolling[i].push(now, row[0])
            self._tip_texts[i] = row[1]
            al = alert_load_for_channel(
                i, row[0], cpu_for_alerts=cpu_for_alerts, meta=meta
            )
            maxed = al >= maxed_frac_for_channel(i)
            self._minute_hot[i].push(now, maxed)
            frac_need = minute_hot_fraction_for_channel(i)
            hot = (
                self._minute_hot[i].sample_count >= MINUTE_HOT_MIN_SAMPLES
                and self._minute_hot[i].fraction_maxed() >= frac_need
            )
            if (
                i == 0
                and hot
                and SUPPRESS_CPU_ALERT_WHEN_FOREGROUND_DOMINATES
                and self._suppress_cpu_hot_due_to_foreground_activity(meta)
            ):
                hot = False
            self._hot_alert[i] = hot

        mean_level = sum(levels) / float(N_METRICS)
        self._rolling_mean.push(now, mean_level)

        focus_idx: int | None
        if self._hover_seg is not None:
            focus_idx = self._hover_seg
            u_single = levels[focus_idx]
        elif self._focus_metric is not None:
            focus_idx = self._focus_metric
            u_single = levels[focus_idx]
        else:
            focus_idx = None
            u_single = mean_level

        per_segment = self._hover_seg is None and self._focus_metric is None
        fh = self._first_hot_index()

        self.canvas.delete("hover_overlay", "focus_overlay", "thanks_overlay", "hot_banner")

        if self._style == "flow":
            if self._flow_fill_id is not None:
                ix1, iy1, ix2, iy2 = self._flow_track
                tw = ix2 - ix1
                if self._hover_seg is not None or self._focus_metric is not None:
                    u = u_single
                else:
                    u = mean_level
                    if self._rolling_mean.spread() >= SPREAD_BAND_FRAC:
                        lo = self._rolling_mean.min() or 0.0
                        hi = self._rolling_mean.max() or 1.0
                        t = math.sin(now * 2.3) * 0.5 + 0.5
                        u = lo + t * (hi - lo)
                if self._hover_seg is not None and self._hot_alert[self._hover_seg]:
                    t = math.sin(now * 3.0) * 0.5 + 0.5
                    fill = self._blend("#ff0000", "#ff6060", t)
                elif (
                    fh is not None
                    and self._hover_seg is None
                    and self._focus_metric is None
                ):
                    t = math.sin(now * 3.0) * 0.5 + 0.5
                    fill = self._blend("#ff0000", "#ff6060", t)
                else:
                    fill = self._fill_blue_red(u)
                w = tw * float(u)
                if u <= 0:
                    self.canvas.coords(self._flow_fill_id, ix1, iy1, ix1, iy2)
                else:
                    self.canvas.coords(
                        self._flow_fill_id,
                        ix1,
                        iy1,
                        ix1 + max(1.0, w),
                        iy2,
                    )
                self.canvas.itemconfigure(self._flow_fill_id, fill=fill)
        else:
            for i, item_ids in enumerate(self._segment_fills):
                fill, outline = self._compute_segment_fill(
                    i, levels, per_segment, u_single, now, pulse_on
                )
                for iid in item_ids:
                    self.canvas.itemconfigure(iid, fill=fill)
                if self._style in ("rhomb", "arrows") and item_ids:
                    pid = item_ids[0]
                    self.canvas.itemconfigure(pid, outline=outline)

        fs = 8 if self._ww >= 200 else 7
        if self._hover_seg is not None:
            hi = self._hover_seg
            med = self._rolling[hi].median()
            pct = f"{med * 100:.0f}%" if med is not None else shorts[hi]
            label_col = "#ff5566" if self._hot_alert[hi] else "#5a6a78"
            pct_col = "#ff9999" if self._hot_alert[hi] else "#8a9cac"
            if self._ww >= 200:
                self.canvas.create_text(
                    8,
                    self._hh / 2,
                    text=METRIC_LABELS[hi],
                    anchor="w",
                    font=("Segoe UI", fs, "bold"),
                    fill=label_col,
                    tags="hover_overlay",
                )
                self.canvas.create_text(
                    self._ww - 6,
                    self._hh / 2,
                    text=pct,
                    anchor="e",
                    font=("Segoe UI", fs + 1),
                    fill=pct_col,
                    tags="hover_overlay",
                )
            else:
                self.canvas.create_text(
                    self._ww - 4,
                    self._hh / 2,
                    text=f"{METRIC_LABELS[hi]} {pct}",
                    anchor="e",
                    font=("Segoe UI", max(6, fs - 1)),
                    fill="#6a7a8a",
                    tags="hover_overlay",
                )
        elif self._focus_metric is not None:
            fi = self._focus_metric
            med = self._rolling[fi].median()
            pct = f"{med * 100:.0f}%" if med is not None else shorts[fi]
            self.canvas.create_text(
                6,
                self._hh - 5,
                text=f"{METRIC_LABELS[fi]}   {pct}",
                anchor="sw",
                font=("Segoe UI", max(6, fs - 1)),
                fill="#3d4a55",
                tags="focus_overlay",
            )
        elif (
            fh is not None
            and self._hover_seg is None
            and self._focus_metric is None
        ):
            self.canvas.create_text(
                8,
                self._hh / 2,
                text=f"{METRIC_LABELS[fh]} !",
                anchor="w",
                font=("Segoe UI", fs, "bold"),
                fill="#ff5566",
                tags="hot_banner",
            )
            self.canvas.create_text(
                self._ww - 8,
                self._hh / 2,
                text="⚠",
                anchor="e",
                font=("Segoe UI", fs + 4),
                fill="#ffaa44",
                tags="hot_banner",
            )
        elif (
            self._thanks_flash_until is not None
            and now < self._thanks_flash_until
        ):
            self.canvas.create_text(
                self._ww - 6,
                self._hh / 2 - 7,
                text="Give",
                anchor="e",
                font=("Segoe UI", max(6, fs - 1)),
                fill="#6a7a8a",
                tags="thanks_overlay",
            )
            self.canvas.create_text(
                self._ww - 6,
                self._hh / 2 + 7,
                text="Thanx",
                anchor="e",
                font=("Segoe UI", max(6, fs - 1)),
                fill="#6a7a8a",
                tags="thanks_overlay",
            )

        self.after(450, self._tick)

    def _start_move(self, e: tk.Event) -> None:
        if self._locked:
            return
        self._tuning_plate = False
        if (
            sys.platform == "win32"
            and self._taskbar_rgb_base is not None
            and (e.state & 0x0001)
        ):
            self._tuning_plate = True
            self._tune_start_y_root = e.y_root
            self._darken_factor_start = self._darken_factor
            return
        self._dx = e.x
        self._dy = e.y

    def _on_move(self, e: tk.Event) -> None:
        if self._tuning_plate and self._taskbar_rgb_base is not None:
            dy = self._tune_start_y_root - e.y_root
            delta = dy / 320.0
            self._darken_factor = max(
                0.65, min(1.0, self._darken_factor_start + delta)
            )
            r, g, b = self._taskbar_rgb_base
            self._plate_bg = rgb_hex_darkened(r, g, b, self._darken_factor)
            self._apply_plate_bgs_to_chrome()
            return
        if self._locked:
            return
        x = e.x_root - self._dx
        y = e.y_root - self._dy
        self.geometry(f"+{x}+{y}")

    def _on_release_move(self, _e: tk.Event) -> None:
        if self._tuning_plate:
            self._tuning_plate = False
        self._persist_config()

            outer,
            text=THANKS_BODIES[variant_ix],
            font=("Segoe UI", 10),
            justify=tk.LEFT,
            wraplength=400,
        )
        body_l.pack(anchor="w", padx=18, pady=(0, 6))

        pick = tk.Frame(outer)
        pick.pack(anchor="w", padx=18, pady=(0, 4))
        tk.Label(pick, text="Presentation:", font=("Segoe UI", 9)).pack(
            side=tk.LEFT, padx=(0, 8)
        )
        om = tk.OptionMenu(
            pick,
            skin_var,
            "Match meter",
            "Whisper",
            "Ink",
            "Paper",
        )
        om.pack(side=tk.LEFT)

        btn_row = tk.Frame(outer)
        btn_row.pack(fill=tk.X, padx=18, pady=(10, 16))

        def another() -> None:
            nonlocal variant_ix
            variant_ix = (variant_ix + 1) % len(THANKS_BODIES)
            body_l.configure(text=THANKS_BODIES[variant_ix])

        another_btn = tk.Button(
            btn_row,
            text="Another voice",
            command=another,
            font=("Segoe UI", 9),
            relief=tk.FLAT,
            padx=10,
            pady=4,
        )
        another_btn.pack(side=tk.LEFT, padx=(0, 8))

        def dismiss() -> None:
            try:
                tw.destroy()
            except tk.TclError:
                pass

        close_btn = tk.Button(
            btn_row,
            text="Close",
            command=dismiss,
            font=("Segoe UI", 9),
            relief=tk.FLAT,
            padx=14,
            pady=4,
        )
        close_btn.pack(side=tk.RIGHT)

        def refresh_theme(*_a: object) -> None:
            c = c_for_theme()
            tw.configure(bg=c["bg"])
            outer.configure(bg=c["bg"])
            pick.configure(bg=c["bg"])
            title_l.configure(fg=c["accent"], bg=c["bg"])
            brand_l.configure(fg=c["muted"], bg=c["bg"])
            sub_l.configure(fg=c["muted"], bg=c["bg"])
            body_l.configure(fg=c["fg"], bg=c["bg"])
            btn_row.configure(bg=c["bg"])
            for w in pick.winfo_children():
                if isinstance(w, tk.Label):
                    w.configure(fg=c["muted"], bg=c["bg"])
            om.configure(
                fg=c["fg"],
                bg=c["bg"],
                activeforeground=c["accent"],
                activebackground=c["bg"],
                highlightbackground=c["accent"],
            )
            another_btn.configure(
                fg=c["muted"],
                bg=c["bg"],
                activeforeground=c["accent"],
                activebackground=c["bg"],
            )
            close_btn.configure(
                fg=c["accent"],
                bg=c["bg"],
                activeforeground=c["fg"],
                activebackground=c["bg"],
            )

        skin_var.trace_add("write", refresh_theme)
        refresh_theme()

        tw.update_idletasks()
        w, h = tw.winfo_reqwidth(), tw.winfo_reqheight()
        px = self.winfo_rootx() + (self._ww - w) // 2
        py = self.winfo_rooty() + (self._hh - h) // 2
        tw.geometry(f"+{max(8, px)}+{max(8, py)}")

    def _menu(self, e: tk.Event) -> None:
        m = tk.Menu(self, tearoff=0)
        if self._locked:
            m.add_command(
                label="Unlock position",
                command=lambda: self._set_locked(False),
            )
        else:
            m.add_command(
                label="Lock position (right-click to unlock)",
                command=lambda: self._set_locked(True),
            )
        m.add_separator()
        m.add_command(
            label="Snap to taskbar band" if self._merge_taskbar else "Snap above taskbar",
            command=lambda: (self._place_window(), self._persist_config()),
        )
        m.add_separator()
        focus_m = tk.Menu(m, tearoff=0)
        focus_m.add_command(
            label="Overview (blend all)",
            command=lambda: self._set_focus_metric(None),
        )
        focus_m.add_separator()
        for i, name in enumerate(METRIC_LABELS):
            focus_m.add_command(
                label=name,
                command=lambda idx=i: self._set_focus_metric(idx),
            )
        m.add_cascade(label="Meter focus", menu=focus_m)
        m.add_separator()
        look = tk.Menu(m, tearoff=0)
        for lbl, st in METER_PRESETS:
            look.add_command(
                label=lbl,
                command=lambda s=st: self._apply_style(s),
            )
        look.add_separator()
        all_styles = tk.Menu(m, tearoff=0)
        for st in STYLES:
            all_styles.add_command(
                label=st,
                command=lambda s=st: self._apply_style(s),
            )
        look.add_cascade(label="All styles…", menu=all_styles)
        m.add_cascade(label="Meter look", menu=look)
        m.add_separator()
        m.add_command(label="About…", command=self._about_dialog)
        if not self._locked:
            m.add_command(label="Give thanx…", command=self._thanks_dialog)
        m.add_command(label="Exit", command=self.destroy)
        try:
            m.tk_popup(e.x_root, e.y_root)
        finally:
            m.grab_release()


def main() -> None:
    p = argparse.ArgumentParser(
        description=(
            f"Lumax stress meter — {BRAND_BYLINE}. Freeware; optional appreciation "
            "(Give thanx, right-click when movable). "
            "Not for sale as commercial screen/taskbar placement (user safety and trust)."
        ),
    )
    p.add_argument(
        "--merge-taskbar",
        action="store_true",
        help="Dock in the Windows taskbar band (next to tray), not inside the shell — but visually merged",
    )
    p.add_argument(
        "--style",
        choices=list(STYLES),
        default=None,
        help=(
            "Visual shell: rhomb | arrows | flow | grille | square | circle | stripe — "
            "omit to use saved style from config"
        ),
    )
    p.add_argument(
        "--opaque",
        action="store_true",
        help="Solid dark background instead of transparent (Windows chroma-key off)",
    )
    p.add_argument(
        "--no-sticky-topmost",
        action="store_true",
        help="Do not periodically re-assert topmost + lift (default: on)",
    )
    p.add_argument(
        "--no-win-topmost",
        action="store_true",
        help="Do not pin HWND_TOPMOST (may slip behind the taskbar; default: on for Windows)",
    )
    p.add_argument(
        "--appbar",
        action="store_true",
        help=(
            "Windows: reserve bottom-edge band via SHAppBarMessage (maximized windows won't cover it). "
            "Must be passed explicitly — not enabled from saved config, for safety."
        ),
    )
    p.add_argument(
        "--no-appbar",
        action="store_true",
        help="Never use AppBar (overrides --appbar). Use to force normal floating window.",
    )
    args = p.parse_args()
    use_appbar = bool(args.appbar) and not args.no_appbar
    cfg = load_meter_config()
    transparent = bool(cfg.get("transparent", True))
    if args.opaque:
        transparent = False
    force_win_topmost = bool(cfg.get("force_win_topmost", sys.platform == "win32"))
    if args.no_win_topmost:
        force_win_topmost = False
    sticky_topmost = bool(cfg.get("sticky_topmost", True))
    if args.no_sticky_topmost:
        sticky_topmost = False
    scfg = cfg.get("style")
    if not isinstance(scfg, str) or scfg not in STYLES:
        scfg = None
    style = args.style if args.style is not None else (scfg or "flow")
    app = StressMeter(
        merge_taskbar=args.merge_taskbar,
        style=style,
        sticky_topmost=sticky_topmost,
        transparent=transparent,
        force_win_topmost=force_win_topmost,
        appbar=use_appbar,
    )
    app.mainloop()


if __name__ == "__main__":
    main()
