    for (const pref of _PREFERRED_VOICE_NAMES) {
        const match = voices.find(v => v.name.includes(pref));
        if (match) {
            (window as any).lumaxPreferredVoice = match;
            localStorage.setItem('lumax_preferred_voice', match.name);
            return;
        }
    }
    // Fallback: pick any English female voice
    const enFemale = voices.find(v => v.lang.startsWith('en') && v.name.toLowerCase().includes('female'));
    if (enFemale) {
        (window as any).lumaxPreferredVoice = enFemale;
        localStorage.setItem('lumax_preferred_voice', enFemale.name);
    }
}
